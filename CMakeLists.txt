cmake_minimum_required(VERSION 3.24)

# GPU architectures: by default the GPU in this machine ("native"; this must be
# set before project(), which otherwise picks the compiler's default, and code
# for an older architecture would be JIT-compiled at every start, slowly).
# For distribution builds pass e.g. -DCMAKE_CUDA_ARCHITECTURES="75;80;86;89;120".
# Compute capability 7.5 (Turing) is the oldest supported target.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT DEFINED ENV{CUDAARCHS})
  set(CMAKE_CUDA_ARCHITECTURES native CACHE STRING "CUDA architectures to build for")
endif()

project(gpuecm2 VERSION 0.1.0 LANGUAGES C CXX CUDA)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# --- dependencies -------------------------------------------------------------
find_package(CUDAToolkit REQUIRED)
# GMP (host only). On Windows use e.g. vcpkg's gmp and point GMP_ROOT at it.
find_path(GMP_INCLUDE_DIR gmp.h HINTS ${GMP_ROOT} PATH_SUFFIXES include)
find_library(GMP_LIBRARY NAMES gmp libgmp mpir HINTS ${GMP_ROOT} PATH_SUFFIXES lib)
if(NOT GMP_INCLUDE_DIR OR NOT GMP_LIBRARY)
  message(FATAL_ERROR "GMP not found; set GMP_ROOT")
endif()
add_library(gmp_lib UNKNOWN IMPORTED)
set_target_properties(gmp_lib PROPERTIES IMPORTED_LOCATION "${GMP_LIBRARY}"
                                         INTERFACE_INCLUDE_DIRECTORIES "${GMP_INCLUDE_DIR}")

# CGBN (header-only, vendored)
add_library(cgbn INTERFACE)
target_include_directories(cgbn SYSTEM INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/cgbn/include)

# --- core library ---------------------------------------------------------------
add_library(gpuecm2core STATIC
  src/host/version.cpp
  src/host/expr.cpp
  src/host/resume.cpp
  src/host/curve.cpp
  src/host/params.cpp
  src/host/roots_plan.cpp
  src/gpu/rns.cu
  src/gpu/ntt.cu
  src/gpu/polymul.cu
  src/gpu/limbops.cu
  src/gpu/polyalg.cu
  src/gpu/roots.cu
  src/gpu/prod.cu
  src/api/stage2.cu
)
# The CGBN kernels (root generation, products) are compiled once per modulus size, each
# size in its own translation unit (gen/roots_<BITS>.cu, gen/prod_<BITS>.cu; see
# src/gpu/cgbn_ops.h), for the sizes of the ladder up to GPUECM2_MAX_BITS: 8192 by default
# (about 15 minutes with -j 2), any size of the ladder up to 32768 on request (the large
# sizes take long to compile, see README.md). GPUECM2_ONLY_BITS compiles a single size,
# used for all moduli up to it (development).
set(GPUECM2_CGBN_LADDER 512 768 1024 1280 1536 1792 2048 2560 3072 3584 4096 4608
                        5120 6144 7168 8192 10240 12288 14336 16384 20480 24576 28672 32768)
set(GPUECM2_MAX_BITS 8192 CACHE STRING "Largest modulus size (bits) to compile CGBN kernels for (up to 32768)")
set(GPUECM2_ONLY_BITS "" CACHE STRING "Compile CGBN kernels for this BITS only (development)")
set(GPUECM2_SIZES "")
if(GPUECM2_ONLY_BITS)
  if(NOT GPUECM2_ONLY_BITS IN_LIST GPUECM2_CGBN_LADDER)
    message(FATAL_ERROR "GPUECM2_ONLY_BITS must be one of ${GPUECM2_CGBN_LADDER}")
  endif()
  set(GPUECM2_SIZES ${GPUECM2_ONLY_BITS})
else()
  foreach(b IN LISTS GPUECM2_CGBN_LADDER)
    if(b LESS_EQUAL GPUECM2_MAX_BITS)
      list(APPEND GPUECM2_SIZES ${b})
    endif()
  endforeach()
  if(NOT GPUECM2_SIZES)
    message(FATAL_ERROR "GPUECM2_MAX_BITS must be at least 512")
  endif()
endif()
list(GET GPUECM2_SIZES -1 GPUECM2_LARGEST)
message(STATUS "gpuecm2: CGBN sizes ${GPUECM2_SIZES} (moduli up to ${GPUECM2_LARGEST} bits)")
set(GPUECM2_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/gen)
set(GPUECM2_SIZE_LIST "")
foreach(b IN LISTS GPUECM2_SIZES)
  string(APPEND GPUECM2_SIZE_LIST " X(${b})")
endforeach()
# Compile order: the root generation of the largest size first (it takes longest:
# 24 minutes and 9 GB of memory at 32768 bits), then the others from large to small,
# but the next two largest (6.4 and 4.6 GB) last, so that with -j 2 the largest
# compiles next to small ones (at most about 11 GB at a time).
set(GPUECM2_SIZES_DOWN ${GPUECM2_SIZES})
list(REVERSE GPUECM2_SIZES_DOWN)
list(LENGTH GPUECM2_SIZES_DOWN GPUECM2_NSIZES)
if(GPUECM2_NSIZES GREATER 3)
  list(GET GPUECM2_SIZES_DOWN 1 GPUECM2_S1)
  list(GET GPUECM2_SIZES_DOWN 2 GPUECM2_S2)
  list(REMOVE_AT GPUECM2_SIZES_DOWN 1 2)
  list(APPEND GPUECM2_SIZES_DOWN ${GPUECM2_S2} ${GPUECM2_S1})
endif()
foreach(part roots prod)
  foreach(b IN LISTS GPUECM2_SIZES_DOWN)
    file(CONFIGURE OUTPUT ${GPUECM2_GEN_DIR}/${part}_${b}.cu
         CONTENT "/* generated by CMake: the CGBN code for ${b}-bit moduli */\n#define GPUECM2_SIZE_BITS ${b}\n#include \"gpu/${part}.cu\"\n"
         @ONLY)
    target_sources(gpuecm2core PRIVATE ${GPUECM2_GEN_DIR}/${part}_${b}.cu)
  endforeach()
endforeach()
file(CONFIGURE OUTPUT ${GPUECM2_GEN_DIR}/gpu/cgbn_size_list.h
     CONTENT "/* generated by CMake: the CGBN sizes of this build (see gpu/cgbn_sizes.h) */\n#pragma once\n#define GPUECM2_CGBN_SIZE_LIST(X)${GPUECM2_SIZE_LIST}\n#define GPUECM2_MAX_BITS ${GPUECM2_LARGEST}\n"
     @ONLY)
target_include_directories(gpuecm2core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src ${GPUECM2_GEN_DIR})
target_link_libraries(gpuecm2core PUBLIC gmp_lib cgbn CUDA::cudart_static)
target_compile_definitions(gpuecm2core PUBLIC GPUECM2_VERSION="${PROJECT_VERSION}")
set_target_properties(gpuecm2core PROPERTIES CUDA_SEPARABLE_COMPILATION OFF
                                             POSITION_INDEPENDENT_CODE ON)

# --- executable -------------------------------------------------------------------
add_executable(gpuecm2 src/cli/main.cpp)
target_link_libraries(gpuecm2 PRIVATE gpuecm2core)

# --- C interface (src/api/gpuecm2.h): libgpuecm2 ----------------------------------
# A shared library exporting only the gpuecm2_* functions (static on Windows).
if(WIN32)
  add_library(gpuecm2lib STATIC src/api/gpuecm2.cpp)
else()
  add_library(gpuecm2lib SHARED src/api/gpuecm2.cpp)
  target_link_options(gpuecm2lib PRIVATE
    "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/api/gpuecm2.map")
  set_target_properties(gpuecm2lib PROPERTIES LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/api/gpuecm2.map
                                              VERSION ${PROJECT_VERSION} SOVERSION 0)
endif()
set_target_properties(gpuecm2lib PROPERTIES OUTPUT_NAME gpuecm2 PUBLIC_HEADER src/api/gpuecm2.h)
target_link_libraries(gpuecm2lib PRIVATE gpuecm2core PUBLIC gmp_lib)
target_include_directories(gpuecm2lib INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)

include(GNUInstallDirs)
install(TARGETS gpuecm2 gpuecm2lib
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        PUBLIC_HEADER DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
